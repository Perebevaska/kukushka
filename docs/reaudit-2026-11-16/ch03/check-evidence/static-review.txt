===== ch03/03b-excel/auto-header =====

import pandas as pd
months = ["Янв", "Фев", "Мар", "Апр", "Май", "Июн", "Июл", "Авг", "Сен"]
book = pd.read_excel("zerno_finance.xlsx", sheet_name=["План", "Факт"], header=None)
heads = {}
sums = {}
for name, sheet in book.items():
    row = sheet[sheet[0] == "Код"].index[0]
    heads[name] = int(row)
    t = pd.read_excel("zerno_finance.xlsx", sheet_name=name, skiprows=row)
    t = t[t["Код"] != "Итого"].set_index("Код")
    sums[name] = t[months].sum(axis=1)
pf = pd.DataFrame({"план": sums["План"], "факт": sums["Факт"]}).round(1)
CHECK
import pandas as _pd
from pathlib import Path as _Path
from io import BytesIO as _BytesIO
import openpyxl as _xl
_m = ['Янв', 'Фев', 'Мар', 'Апр', 'Май', 'Июн', 'Июл', 'Авг', 'Сен']
def _validate(_h, _pf):
    assert isinstance(_h, dict) and set(_h) == {'План', 'Факт'}, 'heads — словарь с ключами «План» и «Факт».'
    _sheets = _pd.read_excel('zerno_finance.xlsx', sheet_name=['План', 'Факт'], header=None)
    _heads = {n: int(t.index[t[0] == 'Код'][0]) for n, t in _sheets.items()}
    assert _h == _heads, f'heads={_h}, по содержимому книги нужно {_heads}. Ищите «Код» отдельно на каждом листе.'
    assert isinstance(_pf, _pd.DataFrame) and list(_pf.columns) == ['план', 'факт'], 'pf — таблица с колонками план и факт.'
    for _col, _name in [('план', 'План'), ('факт', 'Факт')]:
        _t = _pd.read_excel('zerno_finance.xlsx', sheet_name=_name, skiprows=_heads[_name])
        _want = _t[_t['Код'] != 'Итого'].set_index('Код')[_m].sum(axis=1).round(1)
        assert _pf.index.is_unique and set(_pf.index) == set(_want.index), 'Индекс pf — коды кофеен без «Итого».'
        for _key in _want.index:
            assert close(_pf.loc[_key, _col], _want[_key], .051), f'{_col} у {_key}: ожидалось {_want[_key]}.'
_validate(need('heads'), need('pf'))
_path = _Path('zerno_finance.xlsx')
_original = _path.read_bytes()
try:
    _book = _xl.load_workbook(_BytesIO(_original))
    for _name, _n in [('План', 2), ('Факт', 1)]:
        _book[_name].insert_rows(1, _n)
        _book[_name].cell(1, 1, 'Предварительные данные')
    _buf = _BytesIO()
    _book.save(_buf)
    _path.write_bytes(_buf.getvalue())
    _env = {'__name__': '__main__'}
    try:
        exec(compile(_code, '<header-probe>', 'exec'), _env)
    except Exception as _e:
        raise AssertionError(f'После добавления служебных строк код не завершился: {_e}. Ищите заголовок на каждом листе.') from None
    _validate(_env.get('heads'), _env.get('pf'))
finally:
    _path.write_bytes(_original)

import pandas as pd

SYNONYMS = {"соц. сети": "соцсети", "от друзей": "друзья", "проходил мимо": "мимо",
            "2гис": "карты", "яндекс карты": "карты"}

def clean_city(s):
    return (s.str.strip().str.lower().str.removeprefix("г.").str.strip()
             .replace({"екб": "екатеринбург"}).str.capitalize())

def clean_phone(s):
    digits = s.str.replace(r"\D", "", regex=True)
    ok = digits.str.fullmatch(r"[78]?\d{10}")      # 10 цифр или 11 с кодом 7/8
    return digits.where(ok).str[-10:]               # другие длины — пропуск, а не «похожий» номер

def clean_gender(s):
    return s.str.strip().str.lower().map({"м": "М", "муж": "М", "ж": "Ж", "жен": "Ж"})

def parse_birth(s):
    iso = pd.to_datetime(s, format="%Y-%m-%d", errors="coerce")
    ru = pd.to_datetime(s, format="%d.%m.%Y", errors="coerce")
    born = iso.fillna(ru)
    return born.where((2026 - born.dt.year - (born.dt.strftime('%m%d') > '1201')).between(14, 100))

def clean_bonus(s):
    b = pd.to_numeric(s.str.replace(" ", ""), errors="coerce")
    return b.mask((b < 0) | (b > 10_000))

def clean_clients(raw):
    df = raw.drop_duplicates().copy()
    df["client_id"] = df["client_id"].str.strip().str.upper()
    df["name"] = df["name"].str.strip().str.capitalize()
    df["gender"] = clean_gender(df["gender"])
    df["birth_date"] = parse_birth(df["birth_date"])
    df["city"] = clean_city(df["city"])
    df["phone"] = clean_phone(df["phone"])
    df["signup_date"] = pd.to_datetime(df["signup_date"])
    df["bonus"] = clean_bonus(df["bonus"])
    df["source"] = df["source"].str.strip().str.lower().replace(SYNONYMS)
    return (df.sort_values("signup_date")
              .drop_duplicates(subset=["client_id"], keep="last")
              .reset_index(drop=True))

import pandas as pd

SYNONYMS = {"соц. сети": "соцсети", "от друзей": "друзья", "проходил мимо": "мимо",
            "2гис": "карты", "яндекс карты": "карты"}

def clean_city(s):
    return (s.str.strip().str.lower().str.removeprefix("г.").str.strip()
             .replace({"екб": "екатеринбург"}).str.capitalize())

def clean_phone(s):
    digits = s.str.replace(r"\D", "", regex=True)
    ok = digits.str.fullmatch(r"[78]?\d{10}")      # 10 цифр или 11 с кодом 7/8
    return digits.where(ok).str[-10:]               # другие длины — пропуск, а не «похожий» номер

def clean_gender(s):
    return s.str.strip().str.lower().map({"м": "М", "муж": "М", "ж": "Ж", "жен": "Ж"})

def parse_birth(s):
    iso = pd.to_datetime(s, format="%Y-%m-%d", errors="coerce")
    ru = pd.to_datetime(s, format="%d.%m.%Y", errors="coerce")
    born = iso.fillna(ru)
    return born.where((2026 - born.dt.year - (born.dt.strftime('%m%d') > '1201')).between(14, 100))

def clean_bonus(s):
    b = pd.to_numeric(s.str.replace(" ", ""), errors="coerce")
    return b.mask((b < 0) | (b > 10_000))

def clean_clients(raw):
    df = raw.drop_duplicates().copy()
    df["client_id"] = df["client_id"].str.strip().str.upper()
    df["name"] = df["name"].str.strip().str.capitalize()
    df["gender"] = clean_gender(df["gender"])
    df["birth_date"] = parse_birth(df["birth_date"])
    df["city"] = clean_city(df["city"])
    df["phone"] = clean_phone(df["phone"])
    df["signup_date"] = pd.to_datetime(df["signup_date"])
    df["bonus"] = clean_bonus(df["bonus"])
    df["source"] = df["source"].str.strip().str.lower().replace(SYNONYMS)
    return (df.sort_values("signup_date")
              .drop_duplicates(subset=["client_id"], keep="last")
              .reset_index(drop=True))

raw = pd.read_csv("clients_raw.csv")
clients = clean_clients(raw)
log = {"выгрузка": len(raw), "без полных дублей": len(raw.drop_duplicates()), "анкет": len(clients)}

===== ch03/09-project/report =====

SOLUTION
import pandas as pd

SYNONYMS = {"соц. сети": "соцсети", "от друзей": "друзья", "проходил мимо": "мимо",
            "2гис": "карты", "яндекс карты": "карты"}

def clean_city(s):
    return (s.str.strip().str.lower().str.removeprefix("г.").str.strip()
             .replace({"екб": "екатеринбург"}).str.capitalize())

def clean_phone(s):
    digits = s.str.replace(r"\D", "", regex=True)
    ok = digits.str.fullmatch(r"[78]?\d{10}")      # 10 цифр или 11 с кодом 7/8
    return digits.where(ok).str[-10:]               # другие длины — пропуск, а не «похожий» номер

def clean_gender(s):
    return s.str.strip().str.lower().map({"м": "М", "муж": "М", "ж": "Ж", "жен": "Ж"})

def parse_birth(s):
    iso = pd.to_datetime(s, format="%Y-%m-%d", errors="coerce")
    ru = pd.to_datetime(s, format="%d.%m.%Y", errors="coerce")
    born = iso.fillna(ru)
    return born.where((2026 - born.dt.year - (born.dt.strftime('%m%d') > '1201')).between(14, 100))

def clean_bonus(s):
    b = pd.to_numeric(s.str.replace(" ", ""), errors="coerce")
    return b.mask((b < 0) | (b > 10_000))

def clean_clients(raw):
    df = raw.drop_duplicates().copy()
    df["client_id"] = df["client_id"].str.strip().str.upper()
    df["name"] = df["name"].str.strip().str.capitalize()
    df["gender"] = clean_gender(df["gender"])
    df["birth_date"] = parse_birth(df["birth_date"])
    df["city"] = clean_city(df["city"])
    df["phone"] = clean_phone(df["phone"])
    df["signup_date"] = pd.to_datetime(df["signup_date"])
    df["bonus"] = clean_bonus(df["bonus"])
    df["source"] = df["source"].str.strip().str.lower().replace(SYNONYMS)
    return (df.sort_values("signup_date")
              .drop_duplicates(subset=["client_id"], keep="last")
              .reset_index(drop=True))

def report(raw, orders):
    clients = clean_clients(raw)
    stats = orders.groupby("client_id").agg(orders=("order_id", "count"), revenue=("amount", "sum"))
    guests = clients.merge(stats, on="client_id", how="left")
    guests[["orders", "revenue"]] = guests[["orders", "revenue"]].fillna(0)
    age = 2026 - guests["birth_date"].dt.year - (guests["birth_date"].dt.strftime("%m%d") > "1201")
    active = guests[guests["orders"] > 0]
    return {
        "clients": len(guests),
        "active_pct": round((guests["orders"] > 0).mean() * 100, 1),
        "median_age": age.median(),
        "avg_revenue_active": round(active["revenue"].mean(), 1),
    }

raw = pd.read_csv("clients_raw.csv")
orders = pd.read_csv("orders_3m.csv", parse_dates=["datetime"])
report(raw, orders)
