## Воспроизведение проверки итоговых тестов

Команда выполняется из корня репозитория с Python 3. Она читает опубликованные страницы, берёт видимый текст кнопки до пояснения, раскрывает HTML-сущности и нормализует пробелы. При равенстве максимальных длин вопрос не прибавляется к гарантированным верным ответам. Для главы 2 в отдельной колонке учтён список `coreQuestions`, который использует интерфейс.

```python
from html import unescape
import json
from pathlib import Path
import re

course_js = Path("site/assets/course.js").read_text().splitlines()[0]
chapters = json.loads(course_js.split(" = ", 1)[1].rstrip(";"))
for chapter in chapters:
    page = Path("site", chapter["test"]["url"]).read_text()
    questions = re.findall(r'class="quiz-c" data-q="(c\d+)"(.*?</ul>)', page, re.S)
    unique_correct = set()
    for question_id, block in questions:
        options = re.findall(r'<li>\[([ x])\]\s*(.*?)\s*<blockquote>', block, re.S)
        assert len(options) == 4 and sum(mark == "x" for mark, _ in options) == 1
        lengths = []
        for mark, markup in options:
            visible = " ".join(unescape(re.sub(r"<[^>]*>", "", markup)).split())
            lengths.append((mark, len(visible)))
        key = next(length for mark, length in lengths if mark == "x")
        if key > max(length for mark, length in lengths if mark != "x"):
            unique_correct.add(question_id)
    main = set(chapter["test"].get("coreQuestions", [qid for qid, _ in questions]))
    print(chapter["id"], len(unique_correct), len(questions),
          len(unique_correct & main), len(main))
```
